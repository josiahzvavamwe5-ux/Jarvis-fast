import os import google.generativeai as genai import gradio as gr # Secure key from Replit Secrets GOOGLE_API_KEY = os.environ.get("GOOGLE_API_KEY") genai.configure(api_key=GOOGLE_API_KEY) MODEL = "gemini-1.5-flash" # --- IDENTITY SYSTEM (your 5 .md files) --- SOUL = "You are JARVIS V1. Purpose: Helpful, calm, capable. Traits: concise, loyal, secure-by-default." IDENTITY = "You are Manager Agent controlling Research, Coding, Writing specialists." STYLE = "Futuristic, concise, Stark-tech. Use bullets for plans, code blocks for code." MEMORY = [] def research_agent(task): m = genai.GenerativeModel(MODEL, system_instruction=SOUL + " You are Research Agent.") return m.generate_content(f"RESEARCH TASK: {task}. Give clear summary.").text def coding_agent(task): m = genai.GenerativeModel(MODEL, system_instruction=SOUL + " You are Coding Agent. Write clean, commented Python.") return m.generate_content(task).text def writing_agent(task): m = genai.GenerativeModel(MODEL, system_instruction=SOUL + " You are Writing Agent.") return m.generate_content(task).text def planner_agent(task): m = genai.GenerativeModel(MODEL, system_instruction=SOUL + " You are Planner Agent. Break task into steps.") return m.generate_content(f"Create step-by-step plan for: {task}").text def review_agent(draft): m = genai.GenerativeModel(MODEL, system_instruction=SOUL + " You are Review Agent. Critique and improve.") return m.generate_content(f"Review and improve: {draft}").text def manager_route(user_input, history): # Manager decides classifier = genai.GenerativeModel(MODEL) intent = classifier.generate_content( f"Classify '{user_input}' as RESEARCH, CODING, WRITING, PLANNING, or GENERAL. One word only." ).text.strip().upper() print(f"Manager routing -> {intent}") if "RESEARCH" in intent: result = research_agent(user_input) elif "CODING" in intent: result = coding_agent(user_input) elif "WRITING" in intent: result = writing_agent(user_input) elif "PLANNING" in intent: result = planner_agent(user_input) else: base = genai.GenerativeModel(MODEL, system_instruction=SOUL + "\n" + STYLE) result = base.generate_content(user_input).text MEMORY.append({"user": user_input, "jarvis": result}) return result # --- FUTURISTIC UI --- with gr.Blocks(theme=gr.themes.Monochrome(), title="JARVIS V1") as demo: gr.Markdown(""" # J.A.R.V.I.S V1 ### MANAGER → RESEARCH | CODING | WRITING | PLANNER | REVIEW ### MEMORY + IDENTITY SYSTEM ACTIVE """) chatbot = gr.Chatbot(height=450, label="JARVIS Terminal") msg = gr.Textbox(label="Command JARVIS", placeholder="Ex: research quantum computing and create a report") clear = gr.Button("Clear Memory") msg.submit(manager_route, [msg, chatbot], chatbot) msg.submit(lambda: "", None, msg) # clear input def clear_mem(): MEMORY.clear() return [] clear.click(clear_mem, outputs=chatbot) demo.launch()
